

main : Program () () ()
main =
    Platform.worker
        { init = \_ -> ( run results, Cmd.none )
        , update = \_ m -> ( m, Cmd.none )
        , subscriptions = \_ -> Sub.none
        }


run : List ( String, String ) -> ()
run items =
    List.foldl (\( k, v ) _ -> always () (Debug.log k v)) () items
